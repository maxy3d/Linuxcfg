###################
### Install Zsh ###
###################

sudo pacman -S zsh

# Change default Shell

chsh -s /bin/zsh # OR
chsh -s /bin/bash # To go back to bash

#######################
### Install ohMyZsh ###
#######################
#via curl Option 1
sh -c "$(curl -fsSL https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh)"

# or wget Option 2
sh -c "$(wget https://raw.githubusercontent.com/ohmyzsh/ohmyzsh/master/tools/install.sh -O -)"

###################################
### Install Theme powerlevel10K ###
###################################

# Via Oh My Zsh
# Clone the repository directly into your custom themes folder:
git clone --depth=1 https://github.com/romkatv/powerlevel10k.git ${ZSH_CUSTOM:-$HOME/.oh-my-zsh/custom}/themes/powerlevel10k

# Open your ~/.zshrc file and set the theme variable:
ZSH_THEME="powerlevel10k/powerlevel10k"

###################################################
### Install auto-Suggest Plugin Using Oh My Zsh ###
###################################################

# If you already use Oh My Zsh, follow these steps:
# Clone the repository into your Oh My Zsh custom plugins folder:
git clone https://github.com/zsh-users/zsh-autosuggestions ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-autosuggestions

# Open your configuration file using a text editor like nano:
nano ~/.zshrc

# Locate the plugins array and add zsh-autosuggestions inside the parentheses. It should look like this:
plugins=(git zsh-autosuggestions)

# Save and reload your profile:
source ~/.zshrc
