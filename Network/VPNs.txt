#########################
### Install Wireguard ###
#########################

sudo pacman -S wireguard-tools

## use: create or copy config files ##
# Save your provider or server configuration file into /etc/wireguard/wg0.conf.
sudo nano /etc/wireguard/wg0.conf

# Start the connection immediately:
sudo wg-quick up wg0

# Optional: Enable the systemd service to start the VPN automatically at boot:
sudo systemctl enable wg-quick@wg0

# Stop the connection using:
sudo wg-quick down wg0


###############################
